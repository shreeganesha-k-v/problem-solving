Swap adjacent department rows in a table

WITH cte AS (
    SELECT *, ROW_NUMBER() OVER (ORDER BY (SELECT NULL)) AS rn # Select null just ranks every row sequentially
    FROM Employees
)

SELECT *, 
    CASE 
        WHEN rn % 2 = 0 THEN LAG(department, 1, department) OVER (ORDER BY rn) # 3rd param in LAG or LEAD is default value if the lag or lead is missing for that row
        WHEN rn % 2 = 1 THEN LEAD(department, 1, department) OVER (ORDER BY rn)
    END AS newDepartment
FROM cte;

        department,

        DENSE_RANK() OVER (PARTITION BY department ORDER BY salary DESC) AS rn

Find customer's origin to destination
| customerId | origin | dest |
| ---------- | ------ | ---- |
| 1          | A      | B    |
| 1          | B      | C    |
| 2          | X      | Y    |
| 2          | Y      | Z    |

SELECT 
    customerId,
    -- Find origin: origin which is never a destination
    (SELECT origin 
     FROM your_table t2 
     WHERE t1.customerId = t2.customerId 
       AND NOT EXISTS (
           SELECT 1 FROM your_table t3 
           WHERE t3.customerId = t2.customerId AND t3.dest = t2.origin
       )
     LIMIT 1
    ) AS origin,
    
    -- Find destination: dest which is never an origin
    (SELECT dest 
     FROM your_table t2 
     WHERE t1.customerId = t2.customerId 
       AND NOT EXISTS (
           SELECT 1 FROM your_table t3 
           WHERE t3.customerId = t2.customerId AND t3.origin = t2.dest
       )
     LIMIT 1
    ) AS dest
FROM your_table t1
GROUP BY customerId;

Query to find the seats where its previous or current is available along with it. (means if seat 2 is available then return if 1 or 3 is available)

WITH cte AS (
    SELECT 
        seat_id, 
        free,
        -- Check if the current seat and the previous seat are both available
        (lag(free) over (ORDER BY seat_id) * free) AS prev_seat,  
        -- Check if the current seat and the next seat are both available
        (lead(free) over (ORDER BY seat_id) * free) AS next_seat
    FROM Cinema_seat
)

SELECT seat_id
FROM cte
WHERE prev_seat = 1 OR next_seat = 1;

WITH months AS (
    SELECT 2024 AS year, 2 AS month UNION ALL
    SELECT 2024, 3 UNION ALL
    SELECT 2024, 4 UNION ALL
    SELECT 2024, 5
),
product_months AS (
    SELECT p.pname, p.pid, m.year, m.month
    FROM Product p
    CROSS JOIN months m
)
SELECT 
    pm.pname,
    pm.year,
    pm.month,
    COALESCE(SUM(t.amount), 0) AS total_amount
FROM product_months pm
LEFT JOIN Transactions t 
    ON pm.pid = t.pid
    AND YEAR(t.sold_date) = pm.year
    AND MONTH(t.sold_date) = pm.month
GROUP BY pm.pname, pm.year, pm.month
ORDER BY pm.pname, pm.year, pm.month;

Write a SQL query to retrieve the first and last order for each customer from the orders table
| order_id  | customer_id  | order_date  |
| --------- | ------------ | ----------- |
| 1         | 101          | 2024-01-05  |
| 2         | 101          | 2024-03-15  |
| 3         | 101          | 2024-05-20  |
| 4         | 102          | 2024-02-10  |
| 5         | 102          | 2024-04-25  |
| 6         | 102          | 2024-06-30  |
| 7         | 103          | 2024-01-01  |
| 8         | 103          | 2024-02-18  |
| 9         | 103          | 2024-03-25  |
Output:
| customer_id  | first_order_id   | first_order_date   | last_order_id   | last_order_date   |
| ------------ | ---------------- | ------------------ | --------------- | ----------------- |
| 101          | 1                | 2024-01-05         | 3               | 2024-05-20        |
| 102          | 4                | 2024-02-10         | 6               | 2024-06-30        |
| 103          | 7                | 2024-01-01         | 9               | 2024-03-25        |

WITH ranked_orders AS (
  SELECT 
    order_id,
    customer_id,
    order_date,
    RANK() OVER (PARTITION BY customer_id ORDER BY order_date ASC) AS first_rank,
    RANK() OVER (PARTITION BY customer_id ORDER BY order_date DESC) AS last_rank
  FROM orders
)
SELECT 
  customer_id,
  MAX(CASE WHEN first_rank = 1 THEN order_id END) AS first_order_id,
  MAX(CASE WHEN first_rank = 1 THEN order_date END) AS first_order_date,
  MAX(CASE WHEN last_rank = 1 THEN order_id END) AS last_order_id,
  MAX(CASE WHEN last_rank = 1 THEN order_date END) AS last_order_date
FROM ranked_orders
GROUP BY customer_id;

WITH max_in_time AS (
  SELECT emp_id, MAX(time_id) AS in_max
  FROM Office
  WHERE emp_status = 'in'
  GROUP BY emp_id
),
max_out_time AS (
  SELECT emp_id, MAX(time_id) AS out_max
  FROM Office
  WHERE emp_status = 'out'
  GROUP BY emp_id
)

SELECT mi.emp_id
FROM max_in_time mi
LEFT JOIN max_out_time mo ON mi.emp_id = mo.emp_id
WHERE mo.out_max IS NULL OR mi.in_max > mo.out_max;

OR

WITH in_entries AS (
    SELECT * FROM office WHERE emp_status = 'in'
),
out_entries AS (
    SELECT * FROM office WHERE emp_status = 'out'
),
in_out_match AS (
    SELECT 
        i.emp_id,
        i.time_id AS in_time,
        o.time_id AS out_time,
        ROW_NUMBER() OVER (PARTITION BY i.emp_id, i.time_id ORDER BY o.time_id) AS rn
    FROM in_entries i
    LEFT JOIN out_entries o 
        ON i.emp_id = o.emp_id AND o.time_id > i.time_id
)
SELECT emp_id, in_time
FROM in_out_match
WHERE rn = 1 AND out_time IS NULL;
