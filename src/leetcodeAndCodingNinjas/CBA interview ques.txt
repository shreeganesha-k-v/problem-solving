public class FindUnique{

	private List<Integer> method1(List<Integer> nums){
	
		Set<Integer> set = new HashSet<>();
		
		for(int num:nums) set.add(num);
		
		return new ArrayList<>(set);
	}
	
	private List<Integer> method2(List<Integer> nums){
	
		return nums.stream().distinct().collect(Collectors.toList());
	}
	
	private List<Integer> method3(List<Integer> nums){
		List<Integer> res = new ArrayList<>();
		
		for(int num:nums){
			if(!res.contains(num)) res.add(num);
		}
		return res;
		
	}
}
