function secondLargest(arr) {
  let first = -Infinity, second = -Infinity;
  for (let n of arr) {
    if (n > first) [second, first] = [first, n];
    else if (n > second && n != first) second = n;
  }
  return second;
}
console.log(secondLargest([1,5,2,4])); // 4

16)Debounce Function
function debounce(fn, delay) {
  let timer;
  return function (...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), delay);
  };
}

17) Throttle function
function throttle(fn, delay) {
  let last = 0;
  return function (...args) {
    let now = Date.now();
    if (now - last >= delay) {
      last = now;
      fn.apply(this, args);
    }
  };
}

function firstUniqueChar(s) {
  const freq = new Array(26).fill(0);
  const aCode = 'a'.charCodeAt(0);

  // Count frequencies
  for (let char of s) {
    freq[char.charCodeAt(0) - aCode]++;
  }

  // Find first unique character
  for (let i = 0; i < s.length; i++) {
    if (freq[s.charCodeAt(i) - aCode] === 1) {
      return i;
    }
  }

  return -1;
}

function lengthOfLongestSubstring(s) {
    let set = new Set();
    let left = 0, maxLen = 0;

    for (let right = 0; right < s.length; right++) {
        while (set.has(s[right])) {
            set.delete(s[left]);
            left++;
        }
        set.add(s[right]);
        maxLen = Math.max(maxLen, right - left + 1);
    }

    return maxLen;
}

// Example usage:
console.log(lengthOfLongestSubstring("abcabcbb")); // Output: 3
console.log(lengthOfLongestSubstring("bbbbb"));    // Output: 1
console.log(lengthOfLongestSubstring("pwwkew"));   // Output: 3

function rotate(nums, k) {
    k = k % nums.length; // handle k > length

    // Reverse helper
    function reverse(start, end) {
        while (start < end) {
            [nums[start], nums[end]] = [nums[end], nums[start]];
            start++;
            end--;
        }
    }

    // 3-step reverse
    reverse(0, nums.length - 1);       // Reverse whole array
    reverse(0, k - 1);                 // Reverse first k elements
    reverse(k, nums.length - 1);       // Reverse remaining elements
}

Input: [1, 2, 3, 2, 1, 4, 3]
Output: [1, 2, 3, 4]

function uniqueInOrder(arr) {
  const seen = new Set();
  return arr.filter(num => {
    if (!seen.has(num)) {
      seen.add(num);
      return true;
    }
    return false;
  });
}

let sum = 0;

for (const num of arr) {
    sum += num;

    if (map.has(sum - k)) {
        count += map.get(sum - k);

    map.set(sum, (map.get(sum) || 0) + 1);

function isBalanced(s) {
  const stack = [];
  const bracketMap = {
    ')': '(',
    ']': '[',
    '}': '{'
  };

  for (let char of s) {
    if (['(', '[', '{'].includes(char)) {
      stack.push(char);
    } else if ([')', ']', '}'].includes(char)) {
      if (stack.pop() !== bracketMap[char]) {
        return false;
      }
    }
  }

  return stack.length === 0;
}

function longestConsecutive(nums) {
    const numSet = new Set(nums);
    let maxLen = 0;

    for (let num of nums) {
        if (!numSet.has(num - 1)) {
            let currentNum = num;
            let streak = 1;

            while (numSet.has(currentNum + 1)) {
                currentNum++;
                streak++;
            }

            maxLen = Math.max(maxLen, streak);
        }
    }

    return maxLen;
}

function chunkArray(arr, size) {
  if (size <= 0) return []; // handle invalid chunk sizes

  const result = [];

  for (let i = 0; i < arr.length; i += size) {
    const chunk = arr.slice(i, i + size);
    result.push(chunk);
  }

  return result;
}

// Example usage:
console.log(chunkArray([1, 2, 3, 4, 5, 6, 7], 3));
// Output: [[1, 2, 3], [4, 5, 6], [7]]

function majorityElement(nums) {
  let candidate = null;
  let count = 0;

  for (let num of nums) {
    if (count === 0) {
      candidate = num;
    }
    count += (num === candidate) ? 1 : -1;
  }

  return candidate;
}

// Test

function findAnagrams(a, b) {
    let res = [];
    let n = a.length;
    let m = b.length;

    if (n < m) return res;

    let freqA = new Array(26).fill(0);
    let freqB = new Array(26).fill(0);

    for (let char of b) {
        freqB[char.charCodeAt(0) - 97]++;
    }

    for (let j = 0; j < m; j++) {
        freqA[a[j].charCodeAt(0) - 97]++;
    }

    if (isArrayEqual(freqA, freqB)) res.push(0);

    for (let j = m; j < n; j++) {
        freqA[a[j].charCodeAt(0) - 97]++;
        freqA[a[j - m].charCodeAt(0) - 97]--;

        if (isArrayEqual(freqA, freqB)) res.push(j - m + 1);
    }

    return res;
}

function isArrayEqual(a, b) {
    for (let i = 0; i < 26; i++) {
        if (a[i] !== b[i]) return false;
    }
    return true;
}

console.log(findAnagrams('cbaebabacd', 'abc')); // Output: [0, 6]

Longest balanced length
Input: [0, 1, 0, 0, 1, 1, 0]
Output: 6
Explanation: The subarray [1, 0, 0, 1, 1, 0] has equal 0s and 1s.
Solution: convert 0's to -1 so that problem would be to find the longest sequence that retruns 0
function findMaxLength(nums) {
    const map = new Map();
    map.set(0, -1); // sum 0 at index -1 to handle subarrays starting at index 0

    let maxLength = 0;
    let sum = 0;

    for (let i = 0; i < nums.length; i++) {
        sum += nums[i] === 0 ? -1 : 1;

        if (map.has(sum)) {
            maxLength = Math.max(maxLength, i - map.get(sum));
        } else {
            map.set(sum, i);
        }
    }

    return maxLength;
}

// Example:
console.log(findMaxLength([0, 1, 0, 0, 1, 1, 0])); // Output: 6
