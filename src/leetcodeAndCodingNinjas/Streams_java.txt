    .collect(Collectors.groupingBy(Function.identity(), LinkedHashMap::new, Collectors.counting()))
    .entrySet().stream()

public static <T> Collector<T, LinkedList<T>, LinkedList<T>> toLinkedList() {
    return Collector.of(
            LinkedList::new,    // Supplier
            LinkedList::add,    // Accumulator
            (left, right) -> {  // Combiner
                left.addAll(right);
                return left;
            },
            Collector.Characteristics.IDENTITY_FINISH  // Finisher: No conversion needed, identity
    );
}

// Custom Collector to collect sum, count, and average
    public static Collector<Integer, ?, Summary> summarize() {
        return Collector.of(
                Summary::new,  // Supplier: Create a new Summary object
                Summary::add,  // Accumulator: Update sum, count, and average
                Summary::merge, // Combiner: Merge two Summary objects (used in parallel streams)
                Collector.Characteristics.IDENTITY_FINISH // Finisher: No conversion needed, identity
        );
    }

public class LongestConsecutiveGroup {
    public static void main(String[] args) {
        List<Integer> nums = Arrays.asList(100, 4, 200, 1, 3, 2);

        List<Integer> longest = findLongestConsecutive(nums);
        System.out.println(longest);//[1,2,3,4]
    }

    public static List<Integer> findLongestConsecutive(List<Integer> nums) {
        Set<Integer> set = new HashSet<>(nums);

        return nums.stream()
                .filter(n -> !set.contains(n - 1))  // only start from sequence heads
                .map(n -> {
                    int c = n;
                    List<Integer> curr = new ArrayList<>();
                    while (set.contains(c)) {
                        curr.add(c++);
                    }
                    return curr;
                })
                .max(Comparator.comparingInt(List::size))  // return the longest sequence
                .orElse(Collections.emptyList());          // handle Optional properly
    }
}

public class LongestConsecutiveGroup {
    public static void main(String[] args) {
        List<String> words = Arrays.asList("apple", "banana", "apple", "orange", "banana", "apple");
        System.out.println(kMostFrequent(2, words));  // Output: [apple, banana]
    }

    private static List<String> kMostFrequent(int k, List<String> input) {
        return input.stream()
            .collect(Collectors.groupingBy(w -> w, Collectors.counting()))  // <-- fix here
            .entrySet()
            .stream()
            .sorted(Comparator
                .<Map.Entry<String, Long>>comparingLong(Map.Entry::getValue)
                .reversed()
                .thenComparing(Map.Entry::getKey))
            .limit(k)
            .map(Map.Entry::getKey)
            .collect(Collectors.toList());
    }
}

private static Map<String, Long> countMap(List<String> list) {
    return list.stream()
        .collect(Collectors.groupingBy(
            s -> s.chars()
                  .mapToObj(c -> String.valueOf((char) c))
                  .sorted()
                  .collect(Collectors.joining()),
            Collectors.counting()
        ));
}

public class SomeDummy {
    public static void main(String... args) {
        List<Employee> emps = List.of(
            new Employee("AAA", "HR", 300),
            new Employee("SDA", "HR", 400),
            new Employee("ERER", "HR", 250),
            new Employee("John", "IT", 300),
            new Employee("AAA", "IT", 258),
            new Employee("fgfgf", "IT", 255),
            new Employee("aa", "SU", 234)
        );

        Map<String, SummaryStats> statsMap = summaryStats(emps);

        statsMap.entrySet().stream()
            .sorted((e1, e2) -> Double.compare(e2.getValue().avg(), e1.getValue().avg())) // descending by avg
            .forEach(entry -> System.out.println(
                "Dept : " + entry.getKey() + " Stats : [ " + entry.getValue().avg() + " , " + entry.getValue().count() + " ]"
            ));
    }

    public static Map<String, SummaryStats> summaryStats(List<Employee> employees) {
        return employees.stream()
            .collect(Collectors.groupingBy(Employee::dept,
                Collectors.collectingAndThen(
                    Collectors.summarizingInt(Employee::salary),
                    stats -> new SummaryStats(stats.getAverage(), (int) stats.getCount())
                )));
    }

    record Employee(String name, String dept, int salary) {}
    record SummaryStats(Double avg, int count) {}
}
